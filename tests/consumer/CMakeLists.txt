cmake_minimum_required(VERSION 3.20)
project(OpenGCConsumer LANGUAGES C CXX)
set(OPENGC_CONSUMER_PROVIDER "full" CACHE STRING "full or retain")
find_package(OpenGC 0.2 CONFIG REQUIRED)
add_executable(consumer main.c)
set_target_properties(consumer PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
if(OPENGC_CONSUMER_PROVIDER STREQUAL "retain")
    target_compile_definitions(consumer PRIVATE USE_RETAIN=1)
elseif(NOT OPENGC_CONSUMER_PROVIDER STREQUAL "full")
    message(FATAL_ERROR "Unknown provider")
endif()
target_link_libraries(consumer PRIVATE OpenGC::domain OpenGC::allocator OpenGC::${OPENGC_CONSUMER_PROVIDER})
enable_testing()
add_test(NAME installed.consumer COMMAND consumer)
add_executable(heap_consumer ../heap_c.c)
set_target_properties(heap_consumer PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON LINKER_LANGUAGE CXX)
target_link_libraries(heap_consumer PRIVATE OpenGC::abi2_marksweep)
add_test(NAME installed.heap_consumer COMMAND heap_consumer)
add_executable(bump_consumer ../heap_c.c)
target_compile_definitions(bump_consumer PRIVATE OGC_TEST_BUMP=1)
set_target_properties(bump_consumer PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON LINKER_LANGUAGE CXX)
target_link_libraries(bump_consumer PRIVATE OpenGC::abi2_bump)
add_test(NAME installed.bump_consumer COMMAND bump_consumer)

# Identical C archive consumed through installed ABI 3 headers and libraries.
add_library(portable_client STATIC ../portable_client.c)
set_target_properties(portable_client PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
target_link_libraries(portable_client PRIVATE OpenGC::protocol)
foreach(provider IN ITEMS marksweep bump)
    add_executable(portable_${provider} ../portable.cpp)
    target_compile_features(portable_${provider} PRIVATE cxx_std_20)
    target_link_libraries(portable_${provider} PRIVATE portable_client OpenGC::${provider} Threads::Threads)
    add_test(NAME installed.portable_${provider} COMMAND portable_${provider})
endforeach()

# One C99 host drives either isolated policy, with no heap preset dependency.
add_library(policy_client OBJECT ../policy_client.c)
set_target_properties(policy_client PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
target_link_libraries(policy_client PRIVATE OpenGC::protocol)
foreach(provider IN ITEMS marksweep nogc)
    add_executable(policy_${provider} $<TARGET_OBJECTS:policy_client>)
    set_target_properties(policy_${provider} PROPERTIES LINKER_LANGUAGE CXX)
    target_link_libraries(policy_${provider} PRIVATE OpenGC::policy_${provider})
    add_test(NAME installed.policy_${provider} COMMAND policy_${provider})
endforeach()
