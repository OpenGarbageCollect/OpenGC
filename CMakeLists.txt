cmake_minimum_required(VERSION 3.20)
project(OpenGC VERSION 0.2.0 LANGUAGES C CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "OpenGC ABI 1 requires a 64-bit target")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
option(OPENGC_BUILD_TESTS "Build standalone contract tests" ON)
option(OPENGC_BUILD_EXAMPLES "Build the C embedding example" ON)

add_library(opengc_protocol INTERFACE)
add_library(OpenGC::protocol ALIAS opengc_protocol)
set_target_properties(opengc_protocol PROPERTIES EXPORT_NAME protocol)
target_include_directories(opengc_protocol INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)

# Protocol ABI 1 components, isolated policy cores and shared ABI 3 runtime presets.
# A final link selects one policy; ABI 2 compatibility libraries remain explicit.
foreach(component IN ITEMS allocator domain full retain)
    add_library(opengc_${component} STATIC src/${component}.cpp)
    add_library(OpenGC::${component} ALIAS opengc_${component})
    set_target_properties(opengc_${component} PROPERTIES EXPORT_NAME ${component}
        POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
    target_compile_features(opengc_${component} PRIVATE cxx_std_20)
    target_link_libraries(opengc_${component} PUBLIC opengc_protocol)
    if(MSVC)
        target_compile_options(opengc_${component} PRIVATE /W4 /WX /EHs-c-)
    else()
        target_compile_options(opengc_${component} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
    endif()
endforeach()
target_link_libraries(opengc_domain PRIVATE Threads::Threads)
foreach(heap IN ITEMS marksweep bump)
    if(heap STREQUAL "bump")
        set(policy nogc)
    else()
        set(policy marksweep)
    endif()
    add_library(opengc_policy_${policy} STATIC src/${policy}.cpp)
    add_library(OpenGC::policy_${policy} ALIAS opengc_policy_${policy})
    set_target_properties(opengc_policy_${policy} PROPERTIES EXPORT_NAME policy_${policy}
        POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
    target_compile_features(opengc_policy_${policy} PRIVATE cxx_std_20)
    target_link_libraries(opengc_policy_${policy} PUBLIC opengc_protocol)
    add_library(opengc_${heap} STATIC src/heap3_runtime.cpp src/heap3_allocator.cpp)
    add_library(OpenGC::${heap} ALIAS opengc_${heap})
    set_target_properties(opengc_${heap} PROPERTIES EXPORT_NAME ${heap}
        POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
    target_compile_features(opengc_${heap} PRIVATE cxx_std_20)
    if(heap STREQUAL "bump")
        target_compile_definitions(opengc_${heap} PRIVATE OGC_PORTABLE_NOGC=1)
    else()
        target_compile_definitions(opengc_${heap} PRIVATE OGC_PORTABLE_NOGC=0)
    endif()
    target_link_libraries(opengc_${heap} PUBLIC opengc_protocol PRIVATE opengc_policy_${policy} Threads::Threads)
    foreach(target IN ITEMS opengc_policy_${policy} opengc_${heap})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /EHs-c-)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
        endif()
    endforeach()
endforeach()
foreach(heap IN ITEMS marksweep bump)
    # ABI 2 implementations are isolated compatibility libraries. Neither the
    # native ABI 3 provider nor Lisp links them.
    add_library(opengc_abi2_${heap} STATIC src/heap_abi2_${heap}.cpp src/stack_bounds.cpp)
    add_library(OpenGC::abi2_${heap} ALIAS opengc_abi2_${heap})
    set_target_properties(opengc_abi2_${heap} PROPERTIES EXPORT_NAME abi2_${heap}
        POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
    target_compile_features(opengc_abi2_${heap} PRIVATE cxx_std_20)
    target_link_libraries(opengc_abi2_${heap} PUBLIC opengc_protocol PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(opengc_abi2_${heap} PRIVATE /W4 /WX /EHs-c-)
    else()
        target_compile_options(opengc_abi2_${heap} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
    endif()
endforeach()

set(OPENGC_HEAP_PROVIDER "marksweep" CACHE STRING "Default final-link heap provider")
set_property(CACHE OPENGC_HEAP_PROVIDER PROPERTY STRINGS marksweep bump)
if(NOT OPENGC_HEAP_PROVIDER MATCHES "^(marksweep|bump)$")
    message(FATAL_ERROR "Unsupported OPENGC_HEAP_PROVIDER=${OPENGC_HEAP_PROVIDER}")
endif()
add_library(opengc_heap INTERFACE)
add_library(OpenGC::heap ALIAS opengc_heap)
set_target_properties(opengc_heap PROPERTIES EXPORT_NAME heap)
target_link_libraries(opengc_heap INTERFACE opengc_${OPENGC_HEAP_PROVIDER})

if(OPENGC_BUILD_EXAMPLES)
    add_executable(opengc_graph examples/graph.c)
    set_target_properties(opengc_graph PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
    target_link_libraries(opengc_graph PRIVATE opengc_domain opengc_allocator opengc_full)
    add_executable(opengc_graph_cpp examples/graph.cpp)
    target_compile_features(opengc_graph_cpp PRIVATE cxx_std_20)
    target_link_libraries(opengc_graph_cpp PRIVATE opengc_domain opengc_allocator opengc_full)
    if(NOT MSVC)
        target_compile_options(opengc_graph_cpp PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions)
    endif()
endif()
if(OPENGC_BUILD_TESTS)
    enable_testing()
    add_library(opengc_policy_client OBJECT tests/policy_client.c)
    set_target_properties(opengc_policy_client PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
    target_link_libraries(opengc_policy_client PRIVATE opengc_protocol)
    foreach(heap IN ITEMS marksweep nogc)
        add_executable(opengc_policy_${heap}_test $<TARGET_OBJECTS:opengc_policy_client>)
        set_target_properties(opengc_policy_${heap}_test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON LINKER_LANGUAGE CXX)
        target_link_libraries(opengc_policy_${heap}_test PRIVATE opengc_policy_${heap})
        add_test(NAME opengc.policy_${heap} COMMAND opengc_policy_${heap}_test)
    endforeach()
    # The identical precompiled C archive is linked to each provider, never both.
    add_library(opengc_portable_client STATIC tests/portable_client.c)
    target_link_libraries(opengc_portable_client PRIVATE opengc_protocol)
    set_target_properties(opengc_portable_client PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
    foreach(heap IN ITEMS marksweep bump)
        add_executable(opengc_portable_${heap}_test tests/portable.cpp)
        target_compile_features(opengc_portable_${heap}_test PRIVATE cxx_std_20)
        target_link_libraries(opengc_portable_${heap}_test PRIVATE opengc_portable_client opengc_${heap} Threads::Threads)
        add_test(NAME opengc.portable_${heap} COMMAND opengc_portable_${heap}_test)
    endforeach()
    add_executable(opengc_contract tests/contract.cpp)
    target_compile_features(opengc_contract PRIVATE cxx_std_20)
    target_link_libraries(opengc_contract PRIVATE opengc_domain opengc_allocator opengc_full opengc_retain Threads::Threads)
    if(NOT MSVC)
        target_compile_options(opengc_contract PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions)
    endif()
    add_test(NAME opengc.contract COMMAND opengc_contract)
    add_executable(opengc_marksweep_test tests/marksweep.cpp)
    target_compile_features(opengc_marksweep_test PRIVATE cxx_std_20)
    target_link_libraries(opengc_marksweep_test PRIVATE opengc_abi2_marksweep)
    add_test(NAME opengc.marksweep COMMAND opengc_marksweep_test)
    add_executable(opengc_bump_test tests/bump.cpp)
    target_compile_features(opengc_bump_test PRIVATE cxx_std_20)
    target_link_libraries(opengc_bump_test PRIVATE opengc_abi2_bump)
    add_test(NAME opengc.bump COMMAND opengc_bump_test)
    foreach(heap IN ITEMS marksweep bump)
        add_executable(opengc_${heap}_stacks_test tests/stacks.cpp)
        target_compile_features(opengc_${heap}_stacks_test PRIVATE cxx_std_20)
        target_link_libraries(opengc_${heap}_stacks_test PRIVATE opengc_abi2_${heap} Threads::Threads)
        if(heap STREQUAL "bump")
            target_compile_definitions(opengc_${heap}_stacks_test PRIVATE OGC_TEST_BUMP=1)
        endif()
        add_test(NAME opengc.${heap}_stacks COMMAND opengc_${heap}_stacks_test)
    endforeach()
    add_executable(opengc_heap_c_test tests/heap_c.c)
    set_target_properties(opengc_heap_c_test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON LINKER_LANGUAGE CXX)
    target_link_libraries(opengc_heap_c_test PRIVATE opengc_abi2_marksweep)
    add_test(NAME opengc.heap_c COMMAND opengc_heap_c_test)
    add_executable(opengc_bump_c_test tests/heap_c.c)
    target_compile_definitions(opengc_bump_c_test PRIVATE OGC_TEST_BUMP=1)
    set_target_properties(opengc_bump_c_test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON LINKER_LANGUAGE CXX)
    target_link_libraries(opengc_bump_c_test PRIVATE opengc_abi2_bump)
    add_test(NAME opengc.bump_c COMMAND opengc_bump_c_test)
    if(OPENGC_BUILD_EXAMPLES)
        add_test(NAME opengc.c_embedding COMMAND opengc_graph)
        add_test(NAME opengc.cpp_embedding COMMAND opengc_graph_cpp)
    endif()
endif()

install(TARGETS opengc_policy_marksweep opengc_policy_nogc opengc_abi2_marksweep opengc_abi2_bump opengc_heap opengc_protocol opengc_allocator opengc_domain opengc_full opengc_retain opengc_marksweep opengc_bump
    EXPORT OpenGCTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} FILES_MATCHING PATTERN "*.h")
install(EXPORT OpenGCTargets NAMESPACE OpenGC:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OpenGC)
set(OPENGC_DEFAULT_HEAP_PROVIDER "${OPENGC_HEAP_PROVIDER}")
if(DEFINED LISP_GC)
    set(OPENGC_DEFAULT_HEAP_PROVIDER "${LISP_GC}")
endif()
configure_package_config_file(cmake/OpenGCConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/OpenGCConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OpenGC)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/OpenGCConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/OpenGCConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/OpenGCConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OpenGC)
install(FILES COPYING README.md README.zh-CN.md README.ja.md PROTOCOL.md DESIGN.md DEMO.md VALIDATION.md
        DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/OpenGC)
